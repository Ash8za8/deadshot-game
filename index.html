        // --- SOUND ENGINE (Web Audio API Procedural Synth) ---

        class SoundEngine {

            constructor() {

                this.ctx = new (window.AudioContext || window.webkitAudioContext)();

            }


            playGunshot(type) {

                if (this.ctx.state === 'suspended') this.ctx.resume();

                const now = this.ctx.currentTime;

                

                // Noise buffer for gunshot blast

                const bufferSize = this.ctx.sampleRate * 0.2;

                const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);

                const output = buffer.getChannelData(0);

                for (let i = 0; i < bufferSize; i++) {

                    output[i] = Math.random() * 2 - 1;

                }


                const noise = this.ctx.createBufferSource();

                noise.buffer = buffer;


                const filter = this.ctx.createBiquadFilter();

                filter.type = type === 2 ? 'lowpass' : 'bandpass';

                filter.frequency.setValueAtTime(type === 2 ? 800 : 1800, now);


                const gain = this.ctx.createGain();

                gain.gain.setValueAtTime(0.6, now);

                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.18);


                noise.connect(filter);

                filter.connect(gain);

                gain.connect(this.ctx.destination);


                noise.start(now);

            }


            playHitSound() {

                if (this.ctx.state === 'suspended') this.ctx.resume();

                const now = this.ctx.currentTime;

                const osc = this.ctx.createOscillator();

                const gain = this.ctx.createGain();


                osc.type = 'sine';

                osc.frequency.setValueAtTime(1200, now);

                osc.frequency.exponentialRampToValueAtTime(400, now + 0.08);


                gain.gain.setValueAtTime(0.3, now);

                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.08);


                osc.connect(gain);

                gain.connect(this.ctx.destination);


                osc.start(now);

                osc.stop(now + 0.08);

            }


            playFootstep() {

                if (this.ctx.state === 'suspended') this.ctx.resume();

                const now = this.ctx.currentTime;

                const osc = this.ctx.createOscillator();

                const gain = this.ctx.createGain();


                osc.type = 'triangle';

                osc.frequency.setValueAtTime(120, now);

                osc.frequency.exponentialRampToValueAtTime(40, now + 0.05);


                gain.gain.setValueAtTime(0.15, now);

                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.05);


                osc.connect(gain);

                gain.connect(this.ctx.destination);


                osc.start(now);

                osc.stop(now + 0.05);

            }


            playPickupSound() {

                if (this.ctx.state === 'suspended') this.ctx.resume();

                const now = this.ctx.currentTime;

                const osc = this.ctx.createOscillator();

                const gain = this.ctx.createGain();


                osc.type = 'sine';

                osc.frequency.setValueAtTime(523.25, now); // C5

                osc.frequency.setValueAtTime(659.25, now + 0.08); // E5


                gain.gain.setValueAtTime(0.2, now);

                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);


                osc.connect(gain);

                gain.connect(this.ctx.destination);


                osc.start(now);

                osc.stop(now + 0.2);

            }

        // --- WEBRTC MULTIPLAYER NETWORK (PeerJS) ---

        function initNetwork(hostedCode = null) {

            localPlayer.name = nameInput.value.trim() || 'Agent_' + Math.floor(Math.random() * 899 + 100);

            

            updateStatus('Connecting to PeerJS...', 'bg-yellow-500');


            // Initialize PeerJS

            peer = new Peer(hostedCode ? hostedCode : undefined);


            peer.on('open', (id) => {

                localPlayer.id = id;

                roomCode = id;

                displayRoomCode.innerText = id;

                document.getElementById('room-info').classList.remove('hidden');


                if (hostedCode) {

                    isHost = true;

                    updateStatus('Host Active. Waiting for Players...', 'bg-green-500');

                    addChatMessage('System', `Room created! Code: ${id}`);

                } else {

                    updateStatus('Connected to Peer Server', 'bg-green-500');

                }

                

                startMatch();

            });


            peer.on('connection', (conn) => {

                handleConnection(conn);

            });


            peer.on('error', (err) => {

                console.error('PeerJS Error:', err);

                updateStatus('Connection Error: ' + err.type, 'bg-red-500');

            });

        }


        function connectToRoom(code) {

            if (!code) return;

            updateStatus('Connecting to Host...', 'bg-yellow-500');

            

            localPlayer.name = nameInput.value.trim() || 'Agent_' + Math.floor(Math.random() * 899 + 100);

            peer = new Peer();


            peer.on('open', (id) => {

                localPlayer.id = id;

                const conn = peer.connect(code);

                handleConnection(conn);

                startMatch();

            });

        }


        function handleConnection(conn) {

            conn.on('open', () => {

                connections[conn.peer] = conn;

                updateStatus(`Connected to ${Object.keys(connections).length} player(s)`, 'bg-green-500');


                // Send Join Info

                conn.send({

                    type: 'JOIN',

                    id: localPlayer.id,

                    name: localPlayer.name

                });

            });


            conn.on('data', (data) => {

                processNetworkPacket(data, conn.peer);

            });


            conn.on('close', () => {

                if (remotePlayers[conn.peer]) {

                    scene.remove(remotePlayers[conn.peer].mesh);

                    delete remotePlayers[conn.peer];

                }

                delete connections[conn.peer];

                updateStatus(`Players remaining: ${Object.keys(connections).length}`, 'bg-green-500');

            });

        }


        function broadcastData(data) {

            Object.values(connections).forEach(conn => {

                if (conn.open) conn.send(data);

            });

        }


        function processNetworkPacket(data, senderId) {

            switch (data.type) {

                case 'JOIN':

                    createRemotePlayerMesh(senderId, data.name);

                    addChatMessage('System', `${data.name} joined the game.`);

                    break;


                case 'SYNC_STATE':

                    if (!remotePlayers[senderId]) {

                        createRemotePlayerMesh(senderId, data.name || 'Player');

                    }

                    updateRemotePlayerState(senderId, data);

                    break;


                case 'SHOOT':

                    audio.playGunshot(data.weapon);

                    renderBulletTracer(data.origin, data.target);

                    break;


                case 'HIT':

                    if (data.targetId === localPlayer.id) {

                        takeDamage(data.damage, data.attackerName);

                    }

                    break;


                case 'CHAT':

                    addChatMessage(data.sender, data.text);

                    break;


                case 'KILL':

                    addKillfeedEntry(data.killer, data.victim, data.weapon);

                    break;

            }

        }


        // --- REMOTE PLAYER RENDERING ---

        function createRemotePlayerMesh(id, name) {

            const group = new THREE.Group();


            // Low poly player body (Capsule/Box)

            const bodyGeo = new THREE.BoxGeometry(0.8, 1.6, 0.8);

            const bodyMat = new THREE.MeshStandardMaterial({ color: 0xef4444 });

            const body = new THREE.Mesh(bodyGeo, bodyMat);

            body.position.y = 0.8;

            body.castShadow = true;

            group.add(body);


            // Head

            const headGeo = new THREE.BoxGeometry(0.5, 0.5, 0.5);

            const headMat = new THREE.MeshStandardMaterial({ color: 0xfca5a5 });

            const head = new THREE.Mesh(headGeo, headMat);

            head.position.y = 1.7;

            head.castShadow = true;

            group.add(head);


            scene.add(group);


            remotePlayers[id] = {

                id: id,

                name: name,

                mesh: group,

                health: 100,

                kills: 0,

                deaths: 0,

                isBot: false

            };

        }


        function updateRemotePlayerState(id, data) {

            const player = remotePlayers[id];

            if (!player) return;


            // Smooth Interpolation target

            player.mesh.position.set(data.x, data.y - 0.8, data.z);

            player.mesh.rotation.y = data.rotY;

            player.health = data.health;

            player.kills = data.kills;

            player.deaths = data.deaths;

        }


        // --- BOT AI LOGIC (Solo / Auto-Fill) ---

        function spawnBots(count = 3) {

            for (let i = 1; i <= count; i++) {

                const botId = 'bot_' + Math.floor(Math.random() * 10000);

                const botName = 'Bot_' + i;

                

                createRemotePlayerMesh(botId, botName);

                remotePlayers[botId].isBot = true;

                remotePlayers[botId].mesh.position.set((Math.random() - 0.5) * 60, 0, (Math.random() - 0.5) * 60);

                remotePlayers[botId].lastShoot = 0;

            }

        }


        function updateBots(delta) {

            const now = performance.now();


            Object.values(remotePlayers).forEach(player => {

                if (!player.isBot) return;


                const botMesh = player.mesh;

                // Move towards local player

                const dx = localPlayer.x - botMesh.position.x;

                const dz = localPlayer.z - botMesh.position.z;

                const dist = Math.hypot(dx, dz);


                if (dist > 3 && dist < 35) {

                    botMesh.position.x += (dx / dist) * 4 * delta;

                    botMesh.position.z += (dz / dist) * 4 * delta;

                    botMesh.rotation.y = Math.atan2(dx, dz);


                    // Bot Shooting logic

                    if (now - player.lastShoot > 1200) {

                        player.lastShoot = now;

                        audio.playGunshot(1);


                        // Raycast/Line of sight damage check

                        if (dist < 25 && Math.random() < 0.6) {

                            takeDamage(15, player.name);

                        }


                        renderBulletTracer(

                            { x: botMesh.position.x, y: botMesh.position.y + 1.4, z: botMesh.position.z },

                            { x: localPlayer.x, y: localPlayer.y, z: localPlayer.z }

                        );

                    }

                }

            });

        }

        function setupTouchControls() {

            const joystickZone = document.getElementById('touch-joystick-zone');

            const joystickStick = document.getElementById('touch-joystick-stick');

            let touchId = null;

            let startX = 0, startY = 0;


            joystickZone.addEventListener('touchstart', (e) => {

                const t = e.changedTouches[0];

                touchId = t.identifier;

                startX = t.clientX;

                startY = t.clientY;

            });


            joystickZone.addEventListener('touchmove', (e) => {

                for (let i = 0; i < e.changedTouches.length; i++) {

                    const t = e.changedTouches[i];

                    if (t.identifier === touchId) {

                        const dx = t.clientX - startX;

                        const dy = t.clientY - startY;

                        const dist = Math.min(Math.hypot(dx, dy), 40);

                        const angle = Math.atan2(dy, dx);


                        joystickStick.style.transform = `translate(${Math.cos(angle)*dist}px, ${Math.sin(angle)*dist}px)`;


                        // Map to controls

                        keys['KeyW'] = dy < -10;

                        keys['KeyS'] = dy > 10;

                        keys['KeyA'] = dx < -10;

                        keys['KeyD'] = dx > 10;

                    }

                }

            });


            const resetJoystick = () => {

                joystickStick.style.transform = `translate(0px, 0px)`;

                keys['KeyW'] = keys['KeyS'] = keys['KeyA'] = keys['KeyD'] = false;

            };


            joystickZone.addEventListener('touchend', resetJoystick);

        function updatePlayerMovement(delta) {

            const speed = (localPlayer.isCrouching ? 4 : 9) * WEAPONS[localPlayer.currentWeapon].speed;

            

            // Camera Height (Crouch adjustment)

            const targetCamY = localPlayer.isCrouching ? 1.0 : 1.6;

            localPlayer.y += (targetCamY - localPlayer.y) * 0.2;


            // Forward/Right vectors based on rotationY

            const moveZ = (keys['KeyW'] ? 1 : 0) - (keys['KeyS'] ? 1 : 0);

            const moveX = (keys['KeyD'] ? 1 : 0) - (keys['KeyA'] ? 1 : 0);


            if (moveZ !== 0 || moveX !== 0) {

                const dir = new THREE.Vector3(moveX, 0, moveZ).normalize();

                dir.applyAxisAngle(new THREE.Vector3(0, 1, 0), localPlayer.rotationY);


                const nextX = localPlayer.x + dir.x * speed * delta;

                const nextZ = localPlayer.z + dir.z * speed * delta;


                // Simple collision bounding check

                if (!checkMapCollision(nextX, localPlayer.z)) localPlayer.x = nextX;

                if (!checkMapCollision(localPlayer.x, nextZ)) localPlayer.z = nextZ;


                // Footstep sound tick

                if (Math.random() < 0.08) audio.playFootstep();

            }


            // Gravity & Jump Physics

            if (keys['Space'] && !localPlayer.isJumping) {

                localPlayer.vy = 6.5;

                localPlayer.isJumping = true;

            }


            if (localPlayer.isJumping) {

                localPlayer.y += localPlayer.vy * delta;

                localPlayer.vy -= 18 * delta; // Gravity


                const floorY = localPlayer.isCrouching ? 1.0 : 1.6;

                if (localPlayer.y <= floorY) {

                    localPlayer.y = floorY;

                    localPlayer.isJumping = false;

                    localPlayer.vy = 0;

                }

            }


            // Update Three.js Camera orientation

            camera.position.set(localPlayer.x, localPlayer.y, localPlayer.z);

            camera.rotation.order = 'YXZ';

            camera.rotation.y = localPlayer.rotationY;

            camera.rotation.x = localPlayer.rotationX;


            // Health Pickups Collision Check

            healthPickups.forEach(hp => {

                if (hp.active && Math.hypot(localPlayer.x - hp.x, localPlayer.z - hp.z) < 2.0) {

                    hp.active = false;

                    hp.mesh.visible = false;

                    hp.respawnTimer = 15; // 15 seconds cooldown

                    localPlayer.health = Math.min(100, localPlayer.health + 40);

                    audio.playPickupSound();

                    updateHUD();

                }

            });

        }


        function checkMapCollision(px, pz) {

            const playerBox = new THREE.Box3(

                new THREE.Vector3(px - 0.4, 0, pz - 0.4),

                new THREE.Vector3(px + 0.4, 2, pz + 0.4)

            );


            for (let box of mapObstacles) {

                if (playerBox.intersectsBox(box)) return true;

            }

            return false;

        }


        // --- SHOOTING, RECOIL & RAYCASTING ---

        function shootWeapon() {

            const now = performance.now();

            const wpn = WEAPONS[localPlayer.currentWeapon];


            if (now - lastShootTime < wpn.fireRate * 1000) return;

            if (localPlayer.ammo[localPlayer.currentWeapon] <= 0) {

                reloadWeapon();

                return;

            }


            lastShootTime = now;

            localPlayer.ammo[localPlayer.currentWeapon]--;

            updateHUD();


            // Sound

            audio.playGunshot(localPlayer.currentWeapon);


            // Recoil Kick

            localPlayer.rotationX += wpn.recoil;


            // Raycast Shot from Camera Center

            const raycaster = new THREE.Raycaster();

            const screenCenter = new THREE.Vector2(0, 0);


            // Add spread offset

            screenCenter.x += (Math.random() - 0.5) * wpn.spread;

            screenCenter.y += (Math.random() - 0.5) * wpn.spread;


            raycaster.setFromCamera(screenCenter, camera);


            // Bullet Tracer Target

            const rayEnd = raycaster.ray.origin.clone().add(raycaster.ray.direction.clone().multiplyScalar(100));

            renderBulletTracer(camera.position, rayEnd);


            // Broadcast Shoot event

            broadcastData({

                type: 'SHOOT',

                weapon: localPlayer.currentWeapon,

                origin: camera.position,

                target: rayEnd

            });


            // Hit Detection against Remote Players & Bots

            Object.values(remotePlayers).forEach(target => {

                const targetBox = new THREE.Box3().setFromObject(target.mesh);

                const hit = raycaster.ray.intersectBox(targetBox, new THREE.Vector3());


                if (hit) {

                    triggerHitmarker();

                    audio.playHitSound();


                    if (target.isBot) {

                        target.health -= wpn.damage;

                        if (target.health <= 0) {

                            localPlayer.kills++;

                            target.deaths++;

                            addKillfeedEntry(localPlayer.name, target.name, wpn.name);

                            updateHUD();

                            

                            // Respawn bot

                            target.health = 100;

                            target.mesh.position.set((Math.random() - 0.5) * 50, 0, (Math.random() - 0.5) * 50);

                        }

                    } else {

                        // Send Hit packet to networked player

                        if (connections[target.id]) {

                            connections[target.id].send({

                                type: 'HIT',

                                targetId: target.id,

                                damage: wpn.damage,

                                attackerName: localPlayer.name

                            });

                        }

                    }

                }

            });

        }


        function renderBulletTracer(from, to) {

            const points = [new THREE.Vector3(from.x, from.y - 0.2, from.z), new THREE.Vector3(to.x, to.y, to.z)];

            const geo = new THREE.BufferGeometry().setFromPoints(points);

            const mat = new THREE.LineBasicMaterial({ color: 0xfde047, transparent: true, opacity: 0.8 });

            const line = new THREE.Line(geo, mat);

            scene.add(line);


            // Fade out line fast

            setTimeout(() => {

                scene.remove(line);

                geo.dispose();

                mat.dispose();

            }, 60);

        }

                localPlayer.z = (Math.random() - 0.5) * 40;

                broadcastData({

                    type: 'KILL',

                    killer: attackerName,

                    victim: localPlayer.name,

        // --- MAIN GAME LOOP ---

        let lastFrameTime = performance.now();


        function gameLoop(timestamp) {

            requestAnimationFrame(gameLoop);


            const delta = Math.min((timestamp - lastFrameTime) / 1000, 0.1);

            lastFrameTime = timestamp;


            if (!lobbyMenu.classList.contains('hidden')) return;


            // Player logic

            updatePlayerMovement(delta);

            updateBots(delta);


            // Animate Health Pickups rotation & Respawn

            healthPickups.forEach(hp => {

                if (hp.active) {

                    hp.mesh.rotation.y += delta * 2;

                } else {

                    hp.respawnTimer -= delta;

                    if (hp.respawnTimer <= 0) {

                        hp.active = true;

                        hp.mesh.visible = true;

                    }

                }

            });


            // Network Sync Tick (Broadcast state 20 times/sec)

            if (timestamp % 50 < 16) {

                broadcastData({

                    type: 'SYNC_STATE',

                    x: localPlayer.x,

                    y: localPlayer.y,

                    z: localPlayer.z,

                    rotY: localPlayer.rotationY,

                    health: localPlayer.health,

                    kills: localPlayer.kills,

                    deaths: localPlayer.deaths,

                    name: localPlayer.name

                });

            }


            renderer.render(scene, camera);

        }
